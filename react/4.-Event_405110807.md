- **function 정의 대신 =\>** 를 사용할 수 있음
```bash
python
  import logo from './log.svg';
  import './App.css';

  function Header(props) {
    return `<header>`
    <h1>[{
      event.preventDefault(); // 기본 동작 방지. reaload() 방지
      props.onChangeMode();  // 클릭시 이벤트 처리 정의
    }}>{props.title}](/)</h1>

  }
  ...

  function App() {
    // 배열 정의
    const topics = [
      {id:1, title:'html', body: 'html is ...'},
      {id:2, title:'css', body: 'cssis ...'},
      {id:3, title:'javascript', body: 'javascript is ...'}
    ]
    return (

        <Header title="WEB" onChangeMode={()=>{
        alert('Header');}}>`</Header>`  // React function 명은 대문자로 시작
        <Nav topics={topics}></Nav>
        <Article title="Welcome" body="Hello, WEB">`</Article>`
        <Article title="Hi" body="Hello, React">`</Article>`

    );
  }

  export default App;
```

- 전달 값에 대한 이벤트 추가
```bash
python
  import logo from './log.svg';
  import './App.css';

  ...
  function Nav(props) {
    // 출력할 값 정의
    const list = []
    for(let i = 0; i < props.topics.length; i++) {
      let t = props.topics[i];
      list.push(<li key={t.id}>
        <a id={t.id} href={'/read/'+ t.id} onClick={event=>{
        event.preventDefault();
        props.onChangeMode(event.target.id);  // target :  이벤트를 유발시킨 대상(<a>)
      }}>{t.title}</a>)
    }
    return <nav>
    <ol>
      {list}
    </ol>
    </nav>
  }

  function App() {
    // 배열 정의
    const topics = [
      {id:1, title:'html', body: 'html is ...'},
      {id:2, title:'css', body: 'cssis ...'},
      {id:3, title:'javascript', body: 'javascript is ...'}
    ]
    return (

        <Header title="WEB">`</Header>`  // React function 명은 대문자로 시작
        <Nav topics={topics} onChangeMode={(id)=>{
          alert(id);
        }}></Nav>
        <Article title="Welcome" body="Hello, WEB">`</Article>`

    );
  }

  export default App;
```
